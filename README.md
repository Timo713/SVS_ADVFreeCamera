Template for a BepInEx Unity Il2Cpp plugin that uses a `MonoBehaviour`. The template has logging already set up in a way that makes it easy to write log messages from anywhere. It also comes with a toggle for the logs and a configurable LogLevel which can be changed in the plugin config file generated by BepInEx or with the BepInEx ConfigurationManager. Both the plugin and component class use the singleton pattern but you can remove or change it to whatever you need/like.

Based on [SpockBauru BepInEx_IL2CPP_Template](https://github.com/SpockBauru/BepInEx_IL2CPP_Template) and [IllusonMods PluginTemplate](https://github.com/IllusionMods/PluginTemplate)

The plugin and component class with extra details and comments:

#nullable enable
using System.Diagnostics.CodeAnalysis;
using UnityEngine; //Add UnityEngine dependencies from e.g. NuGet
using BepInEx; //Add BepInEx dependencies from e.g. NuGet
using BepInEx.Unity.IL2CPP;
using BepInEx.Configuration;
using BepInEx.Logging;
using HarmonyLib;

using LogLevel = BepInEx.Logging.LogLevel;


//Change namespace and class names to avoid naming conflicts
namespace BepInEx_Il2CppPlugin

	[BepInProcess(PROCESS_NAME)]
	[BepInPlugin(GUID, PLUGIN_NAME, VERSION)]
	public class ExamplePlugin : BasePlugin
	{
		#region PLUGIN_INFO

		/*PLUGIN INFO*/
		//Set the info that is displayed in the plugin DLL properties
		//and used by BepInEx
		public const string PLUGIN_NAME = "PluginName";
		public const string COPYRIGHT = "Copyright ©  2026";
		public const string COMPANY = "LinkOrCompanyName";

		public const string PROCESS_NAME = "ProcessName";
		public const string GUID = "org.PluginGUID";
		public const string VERSION = "1.0.0";

		#endregion



		/*VARIABLES*/
		/// <summary>
		/// Plugin singleton instance. Not set as nullable since it is set in Load(). If you are paranoid null check it
		/// </summary>
		public static ExamplePlugin Instance { get; private set; } = null!;
		
		/// <summary>
		/// Cached ManualLogSource instance for logging
		/// </summary>
		private static ManualLogSource _log = null!;



		/*METHODS*/
		/// <summary>
		/// Returns a bool depending on whether the component in the output parameter is null or not. True == not null
		/// </summary>
		public static bool TryGetComponent([MaybeNullWhen(false)] out ExampleComponent component)
		{
			component = ExampleComponent.Instance;
			return component != null;
		}

		/// <summary>
		/// Gets or adds the component to the BepInEx manager GameObject. Don't Destroy() or make any changes to the GameObject
		/// </summary>
		public static ExampleComponent GetOrAddComponent()
		{
			ExampleComponent? component = ExampleComponent.Instance;

			if (component != null)
			{
				return component;
			}
			else
			{
				return Instance.AddComponent<ExampleComponent>();
			}
		}



		/*CONFIG*/
		//Category which each config entry will be placed under
		public const string CATEGORY_GENERAL = "General";
		public const string CATEGORY_DEBUG = "Debug";

		//General
		public static ConfigEntry<bool> enabled = null!;

		//Debug
		public static ConfigEntry<bool> logEnabled = null!;
		public static ConfigEntry<LogLevel> logLevel = null!;



		/*PLUGIN LOAD*/
		/// <summary>
		/// Is called when the plugin is loaded
		/// </summary>
		public override void Load()
		{
			//Set instance
			Instance = this;
			_log = Log;

			//General
			enabled = Config.Bind(CATEGORY_GENERAL, "Enabled", true); //Parameters are: (string category, string entry name, T default value)

			//Debug
			logEnabled = Config.Bind(CATEGORY_DEBUG, "Enable Logging", false);
            logLevel = Config.Bind(CATEGORY_DEBUG, "Log Levels", LogLevel.Info | LogLevel.Message | LogLevel.Warning | LogLevel.Error | LogLevel.Fatal);

            //Create hooks
            Harmony.CreateAndPatchAll(typeof(ExamplePlugin.Hooks), GUID);
			Harmony.CreateAndPatchAll(typeof(ExampleComponent.Hooks), GUID);
		}

			//Postfix

			public static void Post()

		}



		#region LOGGING

		/* LOGGING

		Log messages with nested Logging class

		In plugin class:
		Logging.LogDebug("Message");
		
		In external class:
		using Logging = BepInEx_Il2CppPlugin.Plugin.Logging;
		Logging.LogDebug("Message");

		Output:
		[Debug  :org.PluginGUID] Message
		
		*/

		public static class Logging
		{
			public static void Log(LogLevel level, string message)
			{
				if (logEnabled.Value && level != LogLevel.None && (logLevel.Value & level) != 0)
				{
					_log.Log(level, message);
				}
			}

			public static void LogFatal(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Fatal) != 0)
				{
					_log.LogFatal(message);
				}
			}

			public static void LogError(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Error) != 0)
				{
					_log.LogError(message);
				}
			}

			public static void LogWarning(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Warning) != 0)
				{
					_log.LogWarning(message);
				}
			}

			public static void LogMessage(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Message) != 0)
				{
					_log.LogMessage(message);
				}
			}

			public static void LogInfo(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Info) != 0)
				{
					_log.LogInfo(message);
				}
			}

			public static void LogDebug(string message)
			{
				if (logEnabled.Value && (logLevel.Value & LogLevel.Debug) != 0)
				{
					_log.LogDebug(message);
				}
			}
		}

		#endregion
	}



	public class ExampleComponent : MonoBehaviour
	{
		/*VARIABLES*/
		/// <summary>
		/// Component singleton instance
		/// </summary>
		public static ExampleComponent? Instance { get; private set; }


		/*INITIALIZATION*/
		private void Start()
		{
			//Initialize component
		}

		private void OnDestroy()
		{
			//Clear instance
			if (Instance == this)
			{
				Instance = null;
			}
		}

		private void Awake()
		{
			//Set instance
			if (Instance == null)
			{
				Instance = this;
			}
			else
			{
				//Call Destroy() on component, not GameObject.
				//Destroying the GameObject will destroy the BepInEx manager GameObject
				//which might affect other plugins
				Destroy(this);
			}
		}
